-- 1. Cabut hak CREATE default pada schema public dari PUBLIC
REVOKE CREATE ON SCHEMA public FROM PUBLIC;
REVOKE
GRANT USAGE ON SCHEMA public TO PUBLIC;
GRANT
-- 2. Buat 3 Role Grup berbasis Least Privilege (NOLOGIN)
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'role_customer_db') THEN
        CREATE ROLE role_customer_db NOLOGIN NOSUPERUSER NOCREATEDB NOCREATEROLE;
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'role_kasir_db') THEN
        CREATE ROLE role_kasir_db NOLOGIN NOSUPERUSER NOCREATEDB NOCREATEROLE;
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'role_auditor_db') THEN
        CREATE ROLE role_auditor_db NOLOGIN NOSUPERUSER NOCREATEDB NOCREATEROLE;
    END IF;
END
$$;
DO
-- Reset privilege lama agar idempoten
REVOKE ALL PRIVILEGES ON ALL TABLES IN SCHEMA public FROM role_customer_db, role_kasir_db, role_auditor_db;
REVOKE
REVOKE ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA public FROM role_customer_db, role_kasir_db, role_auditor_db;
REVOKE
-- 3A. Privilege Minimum untuk role_customer_db (Maker: baca katalog & buat pesanan)
GRANT SELECT ON TABLE catalog_category, catalog_product TO role_customer_db;
GRANT
GRANT SELECT, INSERT ON TABLE rentals_cart, rentals_cart_item, rentals_rental_order, rentals_rental_order_item TO role_customer_db;
GRANT
GRANT USAGE, SELECT ON SEQUENCE rentals_cart_id_seq, rentals_cart_item_id_seq, rentals_rental_order_id_seq, rentals_rental_order_item_id_seq TO role_customer_db;
GRANT
-- 3B. Privilege Minimum untuk role_kasir_db (Checker: verifikasi pesanan, mutasi stok, log inventaris, profil non-password)
GRANT SELECT ON TABLE catalog_category, catalog_product TO role_kasir_db;
GRANT
GRANT SELECT (id, username, first_name, last_name, email, phone_number, is_active) ON TABLE accounts_user TO role_kasir_db;
GRANT
GRANT UPDATE (available_stock, updated_at) ON TABLE catalog_product TO role_kasir_db;
GRANT
GRANT SELECT, UPDATE (
    status, stock_deducted, actual_return_date, overdue_days,
    overdue_fine, damage_fine, total_fine, fine_payment_status,
    return_notes, returned_by_staff_id, updated_at
) ON TABLE rentals_rental_order TO role_kasir_db;
GRANT
GRANT SELECT, UPDATE (condition_at_return, item_damage_fee) ON TABLE rentals_rental_order_item TO role_kasir_db;
GRANT
GRANT SELECT, INSERT ON TABLE catalog_inventory_log, accounts_securityevent TO role_kasir_db;
GRANT
GRANT USAGE, SELECT ON SEQUENCE catalog_inventory_log_id_seq, accounts_securityevent_id_seq TO role_kasir_db;
GRANT
-- 3C. Privilege Minimum untuk role_auditor_db (Read-Only Audit, TANPA kolom password hash)
GRANT SELECT ON TABLE
    accounts_securityevent,
    catalog_inventory_log,
    rentals_rental_order,
    rentals_rental_order_item,
    catalog_category,
    catalog_product,
    auth_group,
    accounts_user_groups
TO role_auditor_db;
GRANT
GRANT SELECT (
    id, username, email, first_name, last_name, phone_number,
    is_active, is_staff, is_superuser, last_login, date_joined,
    failed_login_attempts, locked_until, password_changed_at
) ON TABLE accounts_user TO role_auditor_db;
GRANT
-- 4. Tampilkan ringkasan matriks privilege tabel untuk ketiga role
SELECT grantee, table_name, string_agg(privilege_type, ', ' ORDER BY privilege_type) AS privileges
FROM information_schema.role_table_grants
WHERE grantee IN ('role_customer_db', 'role_kasir_db', 'role_auditor_db')
GROUP BY grantee, table_name
