-- 1. Pasang fungsi & trigger PostgreSQL untuk mencegah Self-Approval (Maker == Checker) pada rentals_rental_order
CREATE OR REPLACE FUNCTION fn_enforce_maker_checker_order()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    -- Cegah pembuat pesanan (user_id) memproses/menyelesaikan pengembalian pesanannya sendiri
    IF NEW.returned_by_staff_id IS NOT NULL
       AND NEW.user_id IS NOT NULL
       AND NEW.returned_by_staff_id = NEW.user_id THEN
        RAISE EXCEPTION 'Maker-Checker Violation: Pembuat pesanan (user_id=%) tidak diizinkan memverifikasi/menyelesaikan pesanannya sendiri (returned_by_staff_id=%).',
            NEW.user_id, NEW.returned_by_staff_id
            USING ERRCODE = '42501';
    END IF;
    RETURN NEW;
END;
$$;
CREATE FUNCTION
DROP TRIGGER IF EXISTS trg_enforce_maker_checker_order ON rentals_rental_order;
DROP TRIGGER
CREATE TRIGGER trg_enforce_maker_checker_order
BEFORE INSERT OR UPDATE ON rentals_rental_order
FOR EACH ROW
EXECUTE FUNCTION fn_enforce_maker_checker_order();
